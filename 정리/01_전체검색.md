# 전체 검색 기능

## 어떤 기능인가?

문서 목록 안에서만 검색하던 것을 벗어나,
**모든 문서의 제목 · 본문 · 요약 · 태그**를 한 번에 검색할 수 있는 페이지입니다.
사이드바 상단 "Search" 링크로 접근하며, URL에 `?q=검색어`가 붙어
뒤로가기를 해도 검색어가 유지됩니다.

## 새로 만든 파일

### `backend/routers/search.py`

```python
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session
from sqlalchemy import or_
from models import Document, Tag, DocumentTag
from routers.auth import get_current_user, get_db
from routers.documents import doc_with_tags
from models import User

router = APIRouter()


@router.get("/search")
def search_documents(
    q: str = Query("", min_length=0),  # q : URL 파라미터 (?q=검색어)
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """
    GET /search?q=검색어

    검색 범위:
      1. Document.title    — 문서 제목
      2. Document.raw_text — 문서 본문
      3. Document.summary  — AI 요약문
      4. Tag.name          — 태그 이름 (해당 태그가 달린 문서 포함)

    반환: doc_with_tags() 형태의 문서 목록 (최신순 정렬)
    """
    if not q.strip():
        return []  # 빈 검색어는 빈 결과 반환

    term = f"%{q.strip()}%"  # SQL LIKE 패턴 — %앞뒤%로 감싸면 부분 일치 검색

    # 1단계: 제목·본문·요약에서 직접 매칭 (ilike = 대소문자 무시)
    text_matches = db.query(Document).filter(
        Document.user_id == current_user.id,
        Document.deleted_at == None,
        or_(
            Document.title.ilike(term),
            Document.raw_text.ilike(term),
            Document.summary.ilike(term),
        ),
    ).all()

    # 2단계: 태그 이름으로 검색 → 해당 태그가 달린 문서 ID 수집
    tag_matched_ids = (
        db.query(DocumentTag.document_id)
        .join(Tag, Tag.id == DocumentTag.tag_id)
        .filter(
            Tag.user_id == current_user.id,
            Tag.name.ilike(term),
        )
        .all()
    )
    tag_doc_ids = {str(row[0]) for row in tag_matched_ids}  # set으로 중복 제거

    tag_matches = []
    if tag_doc_ids:
        tag_matches = db.query(Document).filter(
            Document.user_id == current_user.id,
            Document.deleted_at == None,
            Document.id.in_(tag_doc_ids),
        ).all()

    # 두 결과를 합치고 중복 제거 (같은 문서가 제목·태그 모두 매칭될 수 있음)
    seen: set[str] = set()
    results = []
    for doc in text_matches + tag_matches:
        key = str(doc.id)
        if key not in seen:
            seen.add(key)
            results.append(doc)

    results.sort(key=lambda d: d.created_at, reverse=True)  # 최신순
    return [doc_with_tags(doc, db) for doc in results]
```

### `my-app/src/pages/Search.tsx`

```tsx
// 핵심 로직 요약:
// 1. URL ?q= 파라미터를 초기 검색어로 사용 (링크 공유·뒤로가기 지원)
// 2. Enter 키 또는 검색 버튼으로 API 호출
// 3. highlight() 함수로 결과 텍스트에서 검색어를 <mark>로 강조
// 4. 요약(summary)이 있으면 요약, 없으면 본문 앞 200자를 snippet으로 표시
// 5. 이메일 출처 문서(raw_html 있음)는 "이메일" 배지 표시

const highlight = (text: string) => {
    // 검색어를 정규식으로 분리해서 일치 부분만 <mark> 태그로 감쌈
    const escaped = query.trim().replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    const parts = text.split(new RegExp(`(${escaped})`, "gi"));
    return parts.map((part, i) =>
        part.toLowerCase() === query.trim().toLowerCase()
            ? <mark key={i} className={styles.highlight}>{part}</mark>
            : part
    );
};
```

### `my-app/src/api/search.ts`

```typescript
import { api } from "./client";

// GET /search?q=... — 문서 전체에서 검색
export const searchDocuments = async (q: string) => {
    const res = await api.get("/search", { params: { q } });
    return res.data;
};
```
